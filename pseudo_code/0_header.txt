t_vector
{
	double x;
	double y;
	double z;
}

typedef enum e_type
{
	// SHADOW,
	// BACKGROUND,
	PLANE,
	SPHERE,
	CYLINDER
}	t_type;

t_sphere
{
	t_type		type;
	t_vector	center;
	double		diameter;
	t_vector	color;
}

t_plane
{
	t_type		type;
	t_vector	random_point;
	t_vector	normal_vector;
	t_vector	color;
}

t_cylinder
{
	t_type		type;
	t_vector	center;
	t_vector	axis;
	double		diameter;
	double		height;
	t_vector	color;
}

t_scene
{
	double		ambient_ratio;
	t_vector	color;

	t_vector	camera_coordinates;
	t_vector	camera_normal;
	double		camera_fov;

	t_vector	light_coordinates;
	t_vector	light_color; // =(255,255,255)
	double		light_brightness;

	double		window_width;
	double		window_height;

	double		screen_width;
	double		screen_height;

	t_sphere	*spheres; //list
	t_plane		*planes; //list
	t_cylinder	*cylinders; //list
}

	bool		shadow;
	bool		background;

	t_vector	point_to_light;
	t_vector	color;

	t_scene		scene;
}
