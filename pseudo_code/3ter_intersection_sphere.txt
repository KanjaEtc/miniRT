double	ft_find_ray_sphere_intersection_point(t_scene scene, t_vector from_point,
	t_vector to_point)
{
	t_vector	ray_vector;
	double		quadratic_solutions[2];
	double		a;
	double		b;
	double		c;

	ray_vector = ft_vector_subtraction(to_point, from_point);
	a = ft_scalar(ray_vector, ray_vector);
	b = 2 * ft_scalar(ray_vector, ft_subtraction(from_point, scene.spheres->center));
	c = pow(ft_vector_length(ft_vector_subtraction(from_point, scene.spheres->center)), 2)
		- pow(scene.spheres->diameter / 2);
	return (ft_quadratic_equation_smallest_solution(a, b, c));
}

double	ft_closest_ray_sphere_intersection(t_scene scene, t_pixel *pixel, t_vector from_point,
	t_vector to_point)
{
	double	t;
	double	t_min;
	
	while (scene.spheres)
	{
		t = ft_find_ray_sphere_intersection_point(scene, from_point, to_point);
		if (t > 0 && t < t_min)
		{
			t_min = t;
			pixel->type = SPHERE;
			pixel->object = scene.sphere;
		}
	}
	return (ft_addition(from_point, ft_multiplication(ft_normalize(ft_subtraction(to_point, from_point)), t_min)));
}
