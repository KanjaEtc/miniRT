t_vector	*ft_diffuse_color(t_scene scene, t_pixel *pixel)
{
	t_vector	*diffuse_color;
	double		*light_intensity;
	t_vector	*normal_object;
	t_vector	*normal_light;

	normal_object = ft_find_object_normal_vector(scene, pixel);
	normal_light = ft_normalize(ft_subtraction(scene.light_coordinates, pixel->point_coordinates));
	
	// I = max(0, N . L) * light_brightness
	light_intensity = max(0, ft_scalar(normal_object, normal_light)) * scene.light_brightness;
	
	// Couleur de l'objet * Couleur de la lumiere * Intensite de la lumiere
	diffuse_color = ft_product(pixel->object.color, light_intensity);
	diffuse_color = ft_vector_multiplication(diffuse_color, scene.light_color);

	return (diffuse_color);
}
